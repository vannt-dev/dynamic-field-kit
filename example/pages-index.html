Schema-driven forms

One form schema, four frameworks.

One schema-driven form engine, four frameworks. Describe the fields once as a FieldDescription[] and render them in React, Vue, Angular or Svelte with your own components.

README.md 
import { FieldDescription } from '@dynamic-field-kit/core';

const fields: FieldDescription[] = [
  {
    name: 'username',
    type: 'text',
    label: 'Username',
  },
  {
    name: 'age',
    type: 'number',
    label: 'Age',
    appearCondition: (data) => data.username !== '',
  },
  {
    name: 'greeting',
    type: 'text',
    label: 'Greeting',
    computeValue: (data) => `Hello ${data.username || ''}`.trim(),
  },
];
npm install @dynamic-field-kit/core @dynamic-field-kit/react   # or /vue, /angular, /svelte
